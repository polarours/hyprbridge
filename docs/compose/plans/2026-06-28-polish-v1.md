In `crates/network/src/lib.rs`, replace:

```rust
// Before:
pub struct DiscoveryIndex {
    peers: RwLock<HashMap<String, DiscoveredDevice>>,
}

// After:
use dashmap::DashMap;

pub struct DiscoveryIndex {
    peers: DashMap<String, DiscoveredDevice>,
}
```

Update methods to use DashMap API:

```rust
impl DiscoveryIndex {
    pub async fn upsert_peer(&self, service_fullname: String, peer: DiscoveredDevice) {
        self.peers.insert(service_fullname, peer);
    }

    pub async fn remove_peer(&self, service_fullname: &str) {
        self.peers.remove(service_fullname);
    }

    pub async fn by_device_id(&self, device_id: &DeviceId) -> Option<DiscoveredDevice> {
        self.peers
            .values()
            .find(|peer| &peer.identity.device_id == device_id)
            .cloned()
    }

    pub async fn list(&self) -> Vec<DiscoveredDevice> {
        let mut peers: Vec<DiscoveredDevice> = self.peers.values().cloned().collect();
        peers.sort_by(|left, right| left.identity.name.cmp(&right.identity.name));
        peers
    }
}
```

```rust
use hyprbridge_clipboard::{ClipboardManager, ClipboardPlatform, create_platform_clipboard};

#[tokio::test]
async fn clipboard_manager_poll_detects_changes() {

    // Initial poll should return None (no changes yet)
    let result = manager.poll_once().await;

}

#[tokio::test]
async fn clipboard_content_hash_is_deterministic() {
    use hyprbridge_clipboard::ClipboardContent;

    let content = ClipboardContent {
        content_type: ClipboardContentType::Text,
        data: b"test content".to_vec(),
        file_names: None,
    };

    let hash1 = content.hash();
    let hash2 = content.hash();
    assert_eq!(hash1, hash2);
}

#[tokio::test]
async fn clipboard_history_add_and_retrieve() {
    use hyprbridge_clipboard::{ClipboardHistory, ClipboardHistoryConfig, ClipboardContent};
    use hyprbridge_protocol::ClipboardContentType;

    let mut history = ClipboardHistory::new(ClipboardHistoryConfig::default());
    let content = ClipboardContent {
        content_type: ClipboardContentType::Text,
        data: b"test".to_vec(),
        file_names: None,
    };

    let id = history.add(&content, None);
    assert!(!id.is_empty());
    assert_eq!(history.len(), 1);

    let entry = history.get(&id);
    assert!(entry.is_some());
}
```
